// 🔐 あなたの家計簿アプリ共通のPINコード（パスワード）を設定 const SECRET_PIN = "1234"; function doGet(e) { const p = e.parameter.password; if (p !== SECRET_PIN) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "認証エラー" })).setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const categories = { "支出": [], "収入": [] }; const expenseSetup = ss.getSheetByName("支出設定"); if (expenseSetup) { const vals = expenseSetup.getRange("A1:A" + Math.max(2, expenseSetup.getLastRow())).getValues(); categories["支出"] = vals.map(r => r[0]).filter(v => v !== ""); } const incomeSetup = ss.getSheetByName("収入設定"); if (incomeSetup) { const vals = incomeSetup.getRange("A1:A" + Math.max(2, incomeSetup.getLastRow())).getValues(); categories["収入"] = vals.map(r => r[0]).filter(v => v !== ""); } let records = []; const exHistory = ss.getSheetByName("支出履歴"); if (exHistory && exHistory.getLastRow() > 1) { const rows = exHistory.getRange(2, 1, exHistory.getLastRow() - 1, 5).getValues(); rows.forEach(r => { if(r[0]) { // 🌟【自動補正】「支出履歴」に入っていても、項目が「収入」用なら「収入データ」としてアプリに返します let detectedType = "支出"; if (categories["収入"].includes(r[1])) { detectedType = "収入"; } records.push({ type: detectedType, date: formatDate(r[0]), category: r[1], amount: r[2], location: r[3], notes: r[4], rawDate: new Date(r[0]) }); } }); } const inHistory = ss.getSheetByName("収入履歴"); if (inHistory && inHistory.getLastRow() > 1) { const rows = inHistory.getRange(2, 1, inHistory.getLastRow() - 1, 5).getValues(); rows.forEach(r => { if(r[0]) { // 🌟【自動補正】「収入履歴」に入っていても、項目が「支出」用なら「支出データ」としてアプリに返します let detectedType = "収入"; if (categories["支出"].includes(r[1])) { detectedType = "支出"; } records.push({ type: detectedType, date: formatDate(r[0]), category: r[1], amount: r[2], location: r[3], notes: r[4], rawDate: new Date(r[0]) }); } }); } records.sort((a, b) => b.rawDate - a.rawDate); records = records.slice(0, 50); const output = { status: "success", categories: categories, records: records }; return ContentService.createTextOutput(JSON.stringify(output)).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.password !== SECRET_PIN) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "認証エラー" })).setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActiveSpreadsheet(); // 📸 レシート画像の解析リクエストだった場合の処理 if (data.action === "analyzeReceipt" && data.imageBlob) { return analyzeReceiptOCR(data.imageBlob); } // 🌟【超安全自動振り分けシステム】 // ユーザーがアプリ画面でどちらのボタンを選んでいようと、 // 送られてきた「項目名（カテゴリー）」が「支出設定」と「収入設定」のどちらにあるかで、書き込み先をGASが自動判断します！ let finalType = data.type; // 基本はアプリから送られたもの const expenseSetup = ss.getSheetByName("支出設定"); if (expenseSetup) { const expenseVals = expenseSetup.getRange("A1:A" + Math.max(2, expenseSetup.getLastRow())).getValues().map(r => r[0]); if (expenseVals.includes(data.category)) { finalType = "支出"; } } const incomeSetup = ss.getSheetByName("収入設定"); if (incomeSetup) { const incomeVals = incomeSetup.getRange("A1:A" + Math.max(2, incomeSetup.getLastRow())).getValues().map(r => r[0]); if (incomeVals.includes(data.category)) { finalType = "収入"; } } const sheetName = finalType === "支出" ? "支出履歴" : "収入履歴"; const sheet = ss.getSheetByName(sheetName); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "履歴シートが見つかりません" })).setMimeType(ContentService.MimeType.JSON); } let targetDate = new Date(); if (data.customDate) { targetDate = new Date(data.customDate); } sheet.appendRow([ targetDate, data.category, Number(data.amount), data.location, data.notes ]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // 👁️ Googleドライブの無料OCR機能を使ってレシートをガチで解析する関数 function analyzeReceiptOCR(base64Image) { try { // 1. 送られてきた画像データをGoogleが読める形式に変換 const contentType = base64Image.substring(base64Image.indexOf(":") + 1, base64Image.indexOf(";")); const byteCharacters = Utilities.base64Decode(base64Image.split(",")[1]); const blob = Utilities.newBlob(byteCharacters, contentType, "receipt_temp"); // 2. GoogleドライブのOCR機能（無料）を使って一時的にファイルを作成 const resource = { title: "receipt_ocr_temp", mimeType: contentType }; const tempFile = Drive.Files.insert(resource, blob, { ocr: true, ocrLanguage: "ja" }); // 3. 認識されたテキストを取り出す const doc = DocumentApp.openById(tempFile.id); const text = doc.getBody().getText(); // 4. 使い終わった一時ファイルを即座に完全削除（ドライブを汚しません） Drive.Files.remove(tempFile.id); // 5. テキストから「金額」と「店名」を自動で探し出す let amount = ""; let location = ""; // 金額を探す（「合計」「小計」「税込」などの後ろにある数字、または一番大きい数字など） const lines = text.split("\n"); // 店名は高確率で1行目か2行目にあるので取得 if (lines.length > 0 && lines[0].trim().length > 1) location = lines[0].trim(); if (location.includes("領収") || location.includes("レシート") || location.length < 2) { if (lines.length > 1) location = lines[1].trim(); } // 「合計」の文字が入った行から数字を抜き出す for (let i = 0; i < lines.length; i++) { const line = lines[i].replace(/\s+/g, ""); // スペースを除去 if (line.includes("合計") || line.includes("支払") || line.includes("小計")) { const match = line.match(/[0-9,]{3,7}/); // 3桁〜7桁の数字を探す if (match) { amount = match[0].replace(/,/g, ""); // カンマを消して数字だけにする break; } } } // もし「合計」で見つからなければ、テキスト全体から一番それっぽい数字を探す if (!amount) { const matches = text.match(/[0-9,]{3,6}/g); if (matches) { // レシートの一番下の方に合計が来ることが多いため、後ろから探す for (let j = matches.length - 1; j >= 0; j--) { const num = matches[j].replace(/,/g, ""); if (Number(num) > 10 && Number(num) < 100000) { // 現実的な金額 amount = num; break; } } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", amount: amount || "", location: location.substring(0, 15) || "不明な店舗", // 長すぎたら切る notes: "レシートOCR自動解析" })).setMimeType(ContentService.MimeType.JSON); } catch (e) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "OCR解析失敗: " + e.toString() })).setMimeType(ContentService.MimeType.JSON); } } function formatDate(dateObj) { if (!dateObj) return ""; const d = new Date(dateObj); if (isNaN(d.getTime())) return String(dateObj); const yyyy = d.getFullYear(); const mm = String(d.getMonth() + 1).padStart(2, '0'); const dd = String(d.getDate()).padStart(2, '0'); return `${yyyy}/${mm}/${dd}`; }
